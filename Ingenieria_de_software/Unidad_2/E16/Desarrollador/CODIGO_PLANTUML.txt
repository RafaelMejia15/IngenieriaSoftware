' --- CAPA DE ENTIDADES (Dominio) ---
package "Entities" {
    class Usuario <<entity>> {
        - id : int
        - nombre : String
        - correo : String
    }

    class Rol <<entity>> {
        - id : int
        - nombreRol : String
    }

    class Vacante <<entity>> {
        - id : int
        - titulo : String
        - fechaCierre : Date
        - docsObligatorios : int
    }

    class Candidatura <<entity>> {
        - id : int
        - estado : String
        - fechaPostulacion : Date
    }

    class Expediente <<entity>> {
        - id : int
        - vacante_id : int
        - user_id : String
        - codigo_expediente : String
        - fecha_modifica : String
    }

    class Documento <<entity>> {
        - id : int
        - nombre : String
        - mimeType : String
        - pesoMB : float
        - rutaObjectStore : String
    }

    class Comentario <<entity>> {
        - id : int
        - texto : String
        - fecha : Date
    }

    class Alerta <<entity>> {
        - id : int
        - mensaje : String
        - leida : boolean
    }
}

' Relaciones Estructurales (Dominio)
Usuario "*" -- "1" Rol : "tiene >"
Usuario "1" -- "*" Candidatura : "genera >"
Vacante "1" -- "*" Candidatura : "recibe >"
Candidatura "1" -- "1" Expediente : "posee >"
Expediente "1" *-- "1..*" Documento : "contiene fuerte >"
Documento "1" -- "*" Comentario : "recibe >"
Candidatura "1" -- "*" Alerta : "dispara >"
